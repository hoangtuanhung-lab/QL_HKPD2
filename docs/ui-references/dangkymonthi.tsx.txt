import React, { useState, useMemo, useRef } from 'react';
import { 
    Camera, 
    User, 
    ListChecks, 
    Plus, 
    Save, 
    Edit2, 
    Trash2, 
    Image as ImageIcon,
    AlertCircle
} from 'lucide-react';

const STUDENT_INFO = {
    name: "Nguyễn Văn A",
    level: "THCS",
    school: "TH và THCS Quang Trung",
    dob: "01/01/2012",
    gender: "Nam"
};

// Mock data demonstrating different structures (with/without subgroups) and filtering constraints
const MASTER_SPORTS = [
    {
        id: 'DK',
        name: 'Điền kinh',
        allowedLevels: ['Tiểu học', 'THCS', 'THPT'],
        allowedGenders: ['Nam', 'Nữ'],
        hasSubGroups: true,
        subGroups: [
            {
                id: 'CHAY',
                name: 'Chạy',
                contents: [
                    { id: '100M', name: 'Chạy 100m' },
                    { id: '200M', name: 'Chạy 200m' },
                    { id: '400M', name: 'Chạy 400m' },
                    { id: '1500M', name: 'Chạy 1500m' }
                ]
            },
            {
                id: 'NHAY',
                name: 'Nhảy',
                contents: [
                    { id: 'NX', name: 'Nhảy xa' },
                    { id: 'NC', name: 'Nhảy cao' },
                    { id: 'N3B', name: 'Nhảy ba bước' }
                ]
            }
        ]
    },
    {
        id: 'CV',
        name: 'Cờ vua',
        allowedLevels: ['Tiểu học', 'THCS', 'THPT'],
        allowedGenders: ['Nam', 'Nữ'],
        hasSubGroups: false,
        contents: [
            { id: 'CTC', name: 'Cờ tiêu chuẩn' },
            { id: 'CC', name: 'Cờ chớp' },
            { id: 'CN', name: 'Cờ nhanh' }
        ]
    },
    {
        id: 'BB',
        name: 'Bóng bàn',
        allowedLevels: ['THCS', 'THPT'],
        allowedGenders: ['Nam', 'Nữ'],
        hasSubGroups: false,
        contents: [
            { id: 'DD', name: 'Đơn nam' }, // Realistically we'd filter contents by gender too, but keeping it simple for demo
            { id: 'DDN', name: 'Đơn nữ' },
            { id: 'DDNA', name: 'Đôi nam' }
        ]
    },
    {
        id: 'TDNT',
        name: 'Thể dục nhịp điệu',
        allowedLevels: ['THCS', 'THPT'],
        allowedGenders: ['Nữ'], // Only females, should be filtered out for this user
        hasSubGroups: false,
        contents: [
            { id: 'BD', name: 'Bài tự chọn' }
        ]
    }
];

export default function App() {
    const [photo, setPhoto] = useState(null);
    const fileInputRef = useRef(null);
    
    // Initial state: 1 empty row in edit mode
    const [registrations, setRegistrations] = useState([
        { id: 1, monId: '', subGroupId: '', contentId: '', isEditing: true }
    ]);
    const [nextId, setNextId] = useState(2);
    const [errorMsg, setErrorMsg] = useState('');

    // Filter sports based on student info
    const availableSports = useMemo(() => {
        return MASTER_SPORTS.filter(sport => 
            sport.allowedLevels.includes(STUDENT_INFO.level) &&
            sport.allowedGenders.includes(STUDENT_INFO.gender)
        );
    }, []);

    const handlePhotoUpload = (e) => {
        const file = e.target.files[0];
        if (file) {
            const reader = new FileReader();
            reader.onload = (event) => setPhoto(event.target.result);
            reader.readAsDataURL(file);
        }
    };

    const generateMaMon = (monId, subGroupId, contentId) => {
        if (!monId) return "---";
        let code = monId;
        code += subGroupId ? `-${subGroupId}` : `-XX`;
        code += contentId ? `-${contentId}` : `-XXX`;
        return code;
    };

    const updateRow = (id, field, value) => {
        setRegistrations(prev => prev.map(row => {
            if (row.id === id) {
                const newRow = { ...row, [field]: value };
                // Reset cascading fields
                if (field === 'monId') {
                    newRow.subGroupId = '';
                    newRow.contentId = '';
                }
                if (field === 'subGroupId') {
                    newRow.contentId = '';
                }
                return newRow;
            }
            return row;
        }));
        setErrorMsg(''); // Clear errors on change
    };

    const toggleEdit = (id, isEditing) => {
        setRegistrations(prev => prev.map(row => 
            row.id === id ? { ...row, isEditing } : row
        ));
    };

    const saveRow = (id) => {
        const row = registrations.find(r => r.id === id);
        const sport = availableSports.find(s => s.id === row.monId);
        
        if (!sport) {
            setErrorMsg("Vui lòng chọn Môn thi đấu.");
            return;
        }
        if (sport.hasSubGroups && !row.subGroupId) {
            setErrorMsg("Vui lòng chọn Phân nhóm.");
            return;
        }
        if (!row.contentId) {
            setErrorMsg("Vui lòng chọn Nội dung thi đấu.");
            return;
        }

        toggleEdit(id, false);
        setErrorMsg('');
    };

    const deleteRow = (id) => {
        setRegistrations(prev => prev.filter(row => row.id !== id));
    };

    const addEmptyRow = () => {
        setRegistrations(prev => [
            ...prev,
            { id: nextId, monId: '', subGroupId: '', contentId: '', isEditing: true }
        ]);
        setNextId(nextId + 1);
    };

    const addRowWithCopiedSubject = () => {
        const lastRow = registrations[registrations.length - 1];
        if (lastRow && lastRow.monId) {
            setRegistrations(prev => [
                ...prev,
                { 
                    id: nextId, 
                    monId: lastRow.monId, 
                    subGroupId: lastRow.subGroupId, 
                    contentId: '', // Leave content empty for new selection
                    isEditing: true 
                }
            ]);
        } else {
            // Fallback to empty row if no previous valid row
            addEmptyRow();
        }
        setNextId(nextId + 1);
    };

    return (
        <div className="min-h-screen bg-slate-50 p-4 md:p-8 font-sans text-slate-800">
            <div className="max-w-6xl mx-auto space-y-6">
                
                {/* Header */}
                <header className="bg-white p-6 rounded-2xl shadow-sm border border-slate-200 flex flex-col items-center justify-center text-center">
                    <h1 className="text-2xl md:text-3xl font-bold text-indigo-700">Đăng ký môn thi đấu Hội khỏe Phù Đổng xã Quảng Phú</h1>
                    <div className="mt-3 text-sm font-medium text-rose-600 bg-rose-50 px-4 py-1.5 rounded-full border border-rose-100 flex items-center gap-2">
                        <AlertCircle className="w-4 h-4" /> Thời hạn đăng ký: 15/10/2026
                    </div>
                </header>

                {/* Phần 1: Thông tin chung */}
                <section className="bg-white rounded-2xl shadow-sm border border-slate-200 p-6">
                    <h2 className="text-lg font-bold text-slate-700 border-b pb-3 mb-6 flex items-center gap-2">
                        <User className="w-5 h-5 text-indigo-600" />
                        Phần 1: Thông tin chung
                    </h2>
                    
                    <div className="flex flex-col md:flex-row gap-8 items-start">
                        {/* 4x6 Photo Upload */}
                        <div className="flex-shrink-0 w-full md:w-auto flex flex-col items-center">
                            <div 
                                onClick={() => fileInputRef.current?.click()}
                                className="relative group cursor-pointer w-[120px] h-[160px] bg-slate-100 border-2 border-dashed border-slate-300 rounded-xl flex flex-col items-center justify-center overflow-hidden hover:border-indigo-400 hover:bg-indigo-50 transition-colors"
                            >
                                {photo ? (
                                    <img src={photo} alt="Ảnh 4x6" className="absolute inset-0 w-full h-full object-cover z-10" />
                                ) : (
                                    <div className="text-center p-4 z-0 flex flex-col items-center group-hover:text-indigo-600">
                                        <ImageIcon className="w-8 h-8 mb-2 text-slate-400 group-hover:text-indigo-500" />
                                        <span className="text-xs font-medium text-slate-500 group-hover:text-indigo-600">Tải ảnh lên</span>
                                        <span className="text-[10px] text-slate-400 mt-1">4x6 cm</span>
                                    </div>
                                )}
                                {/* Overlay on hover if photo exists */}
                                {photo && (
                                    <div className="absolute inset-0 bg-black/40 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity z-20">
                                        <Camera className="w-6 h-6 text-white" />
                                    </div>
                                )}
                            </div>
                            <input 
                                type="file" 
                                ref={fileInputRef}
                                onChange={handlePhotoUpload} 
                                accept="image/*" 
                                className="hidden" 
                            />
                        </div>

                        {/* Readonly Info */}
                        <div className="flex-grow w-full">
                            <div className="grid grid-cols-1 sm:grid-cols-2 gap-x-8 gap-y-6 bg-slate-50 p-6 rounded-xl border border-slate-100">
                                <div className="space-y-1">
                                    <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Họ và tên</span>
                                    <div className="text-base font-medium text-slate-900">{STUDENT_INFO.name}</div>
                                </div>
                                <div className="space-y-1">
                                    <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Ngày sinh</span>
                                    <div className="text-base font-medium text-slate-900">{STUDENT_INFO.dob}</div>
                                </div>
                                <div className="space-y-1">
                                    <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Giới tính</span>
                                    <div className="text-base font-medium text-slate-900">{STUDENT_INFO.gender}</div>
                                </div>
                                <div className="space-y-1">
                                    <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Cấp học</span>
                                    <div className="text-base font-medium text-slate-900">{STUDENT_INFO.level}</div>
                                </div>
                                <div className="space-y-1 sm:col-span-2">
                                    <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Tên trường</span>
                                    <div className="text-base font-medium text-slate-900">{STUDENT_INFO.school}</div>
                                </div>
                            </div>
                        </div>
                    </div>
                </section>

                {}
                <section className="bg-white rounded-2xl shadow-sm border border-slate-200 p-6">
                    <h2 className="text-lg font-bold text-slate-700 border-b pb-3 mb-4 flex items-center gap-2">
                        <ListChecks className="w-5 h-5 text-indigo-600" />
                        Phần 2: Danh sách đăng ký
                    </h2>

                    {errorMsg && (
                        <div className="mb-4 p-3 bg-red-50 border border-red-200 text-red-600 rounded-lg text-sm flex items-center gap-2">
                            <AlertCircle className="w-4 h-4" /> {errorMsg}
                        </div>
                    )}

                    <div className="overflow-x-auto rounded-xl border border-slate-200 mb-4">
                        <table className="w-full text-left border-collapse whitespace-nowrap min-w-[800px]">
                            <thead>
                                <tr className="bg-slate-50 text-slate-600 text-sm font-semibold border-b border-slate-200">
                                    <th className="py-3 px-4 w-12 text-center">STT</th>
                                    <th className="py-3 px-4 w-32 text-center">Mã môn</th>
                                    <th className="py-3 px-4 w-48">Môn</th>
                                    <th className="py-3 px-4 w-48">Phân nhóm</th>
                                    <th className="py-3 px-4 min-w-[200px]">Tên nội dung</th>
                                    <th className="py-3 px-4 w-40 text-center">Trạng thái</th>
                                </tr>
                            </thead>
                            <tbody className="text-sm align-middle">
                                {registrations.length === 0 ? (
                                    <tr>
                                        <td colSpan="6" className="py-8 text-center text-slate-400">
                                            Chưa có dữ liệu đăng ký.
                                        </td>
                                    </tr>
                                ) : (
                                    registrations.map((row, index) => {
                                        const sport = availableSports.find(s => s.id === row.monId);
                                        const subGroup = sport?.subGroups?.find(sg => sg.id === row.subGroupId);
                                        
                                        // Determine contents list
                                        let contents = [];
                                        if (sport) {
                                            if (sport.hasSubGroups) {
                                                contents = subGroup ? subGroup.contents : [];
                                            } else {
                                                contents = sport.contents;
                                            }
                                        }

                                        const code = generateMaMon(row.monId, row.subGroupId, row.contentId);

                                        return (
                                            <tr key={row.id} className="border-b border-slate-100 hover:bg-slate-50/50">
                                                <td className="py-3 px-4 text-center font-medium text-slate-500">
                                                    {index + 1}
                                                </td>
                                                <td className="py-3 px-4 text-center">
                                                    <span className="font-mono text-xs font-semibold text-indigo-700 bg-indigo-50 px-2 py-1 rounded">
                                                        {code}
                                                    </span>
                                                </td>

                                                {/* Edit Mode vs View Mode */}
                                                {row.isEditing ? (
                                                    <>
                                                        <td className="py-2 px-3">
                                                            <select 
                                                                value={row.monId}
                                                                onChange={(e) => updateRow(row.id, 'monId', e.target.value)}
                                                                className="w-full border border-slate-300 rounded-lg p-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 outline-none"
                                                            >
                                                                <option value="">-- Chọn Môn --</option>
                                                                {availableSports.map(s => (
                                                                    <option key={s.id} value={s.id}>{s.name}</option>
                                                                ))}
                                                            </select>
                                                        </td>
                                                        <td className="py-2 px-3">
                                                            <select 
                                                                value={row.subGroupId}
                                                                onChange={(e) => updateRow(row.id, 'subGroupId', e.target.value)}
                                                                disabled={!sport || !sport.hasSubGroups}
                                                                className="w-full border border-slate-300 rounded-lg p-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 outline-none disabled:bg-slate-100 disabled:text-slate-400"
                                                            >
                                                                <option value="">-- Chọn Nhóm --</option>
                                                                {sport?.hasSubGroups && sport.subGroups.map(sg => (
                                                                    <option key={sg.id} value={sg.id}>{sg.name}</option>
                                                                ))}
                                                            </select>
                                                        </td>
                                                        <td className="py-2 px-3">
                                                            <select 
                                                                value={row.contentId}
                                                                onChange={(e) => updateRow(row.id, 'contentId', e.target.value)}
                                                                disabled={!sport || (sport.hasSubGroups && !row.subGroupId)}
                                                                className="w-full border border-slate-300 rounded-lg p-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 outline-none disabled:bg-slate-100 disabled:text-slate-400"
                                                            >
                                                                <option value="">-- Chọn Nội dung --</option>
                                                                {contents.map(c => (
                                                                    <option key={c.id} value={c.id}>{c.name}</option>
                                                                ))}
                                                            </select>
                                                        </td>
                                                        <td className="py-2 px-4 text-center">
                                                            <div className="flex items-center justify-center gap-2">
                                                                <button onClick={() => saveRow(row.id)} className="p-1.5 text-emerald-600 hover:bg-emerald-50 rounded transition-colors" title="Lưu">
                                                                    <Save className="w-5 h-5" />
                                                                </button>
                                                                <button onClick={() => deleteRow(row.id)} className="p-1.5 text-red-500 hover:bg-red-50 rounded transition-colors" title="Xóa">
                                                                    <Trash2 className="w-5 h-5" />
                                                                </button>
                                                            </div>
                                                        </td>
                                                    </>
                                                ) : (
                                                    <>
                                                        <td className="py-3 px-4 text-slate-800">{sport?.name || '-'}</td>
                                                        <td className="py-3 px-4 text-slate-600">{subGroup?.name || '-'}</td>
                                                        <td className="py-3 px-4 text-slate-800 font-medium">
                                                            {contents.find(c => c.id === row.contentId)?.name || '-'}
                                                        </td>
                                                        <td className="py-3 px-4 text-center">
                                                            <div className="flex items-center justify-center gap-2">
                                                                <button onClick={() => toggleEdit(row.id, true)} className="p-1.5 text-indigo-600 hover:bg-indigo-50 rounded transition-colors" title="Sửa">
                                                                    <Edit2 className="w-4 h-4" />
                                                                </button>
                                                                <button onClick={() => deleteRow(row.id)} className="p-1.5 text-red-500 hover:bg-red-50 rounded transition-colors" title="Xóa">
                                                                    <Trash2 className="w-4 h-4" />
                                                                </button>
                                                            </div>
                                                        </td>
                                                    </>
                                                )}
                                            </tr>
                                        );
                                    })
                                )}
                            </tbody>
                        </table>
                    </div>

                    {}
                    <div className="flex items-center gap-6 mt-2 px-2">
                        <button 
                            onClick={addEmptyRow}
                            className="flex items-center gap-1.5 text-indigo-600 hover:text-indigo-800 font-medium text-sm transition-colors group"
                        >
                            <Plus className="w-4 h-4 group-hover:scale-110 transition-transform" />
                            Thêm môn thi
                        </button>
                        
                        <button 
                            onClick={addRowWithCopiedSubject}
                            className="flex items-center gap-1.5 text-emerald-600 hover:text-emerald-800 font-medium text-sm transition-colors group"
                            title="Thêm nội dung dựa trên môn học/phân nhóm của dòng cuối cùng"
                        >
                            <Plus className="w-4 h-4 group-hover:scale-110 transition-transform" />
                            Thêm nội dung
                        </button>
                    </div>
                </section>
            </div>
        </div>
    );
}
